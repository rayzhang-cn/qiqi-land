```tsx
import { useState } from 'react';
import type { ClassifyData } from '../content/schema';

interface Props {
  data: ClassifyData;
  onCorrect: () => void;
  onMistake: () => void;
}

export function ClassifyGame({ data, onCorrect, onMistake }: Props) {
  const [selected, setSelected] = useState<string | null>(null);
  const [placed, setPlaced] = useState<Record<string, boolean>>({});
  const [wrong, setWrong] = useState<string | null>(null);
  const [solved, setSolved] = useState(false);

  const remaining = data.items.filter((it) => !placed[it.id]);

  const handleSelect = (itemId: string) => {
    if (solved) return;
    setSelected(itemId);
    setWrong(null);
  };

  const handleBucket = (bucketId: string) => {
    if (solved || !selected) return;
    const item = data.items.find((it) => it.id === selected);
    if (!item) return;
    if (item.bucketId === bucketId) {
      setPlaced((p) => ({ ...p, [item.id]: true }));
      setSelected(null);
      setWrong(null);
      if (remaining.length === 1) {
        setSolved(true);
        onCorrect();
      }
    } else {
      setWrong(selected);
      onMistake();
    }
  };

  return (
    <div className="classify">
      <div className="classify-buckets">
        {data.buckets.map((b) => (
          <button key={b.id} className="classify-bucket" onClick={() => handleBucket(b.id)} aria-label={b.label}>
            <span className="classify-bucket-emoji">{b.emoji}</span>
            <span>{b.label}</span>
          </button>
        ))}
      </div>
      <div className="classify-items">
        {data.items.map((it) =>
          placed[it.id] ? (
            <div key={it.id} className="classify-item classify-item-placed" />
          ) : (
            <button
              key={it.id}
              className={`classify-item ${selected === it.id ? 'classify-item-selected' : ''} ${wrong === it.id ? 'shake' : ''}`}
              onClick={() => handleSelect(it.id)}
              aria-label={it.label}
            >
              <span className="classify-item-emoji">{it.emoji}</span>
            </button>
          ),
        )}
      </div>
    </div>
  );
}
```

```tsx
import { useRef, useState } from 'react';
import type { ColoringData } from '../content/schema';

interface Props {
  data: ColoringData;
  onCorrect: () => void;
}

interface Segment {
  color: string;
  x1: number;
  y1: number;
  x2: number;
  y2: number;
  width: number;
}

export function ColoringGame({ data, onCorrect }: Props) {
  const [color, setColor] = useState(data.palette[0]);
  const [isEraser, setIsEraser] = useState(false);
  const [segments, setSegments] = useState<Segment[]>([]);
  const drawing = useRef(false);
  const last = useRef<{ x: number; y: number } | null>(null);
  const svgRef = useRef<SVGSVGElement | null>(null);

  const strokeColor = isEraser ? '#ffffff' : color;
  const strokeWidth = isEraser ? 24 : 12;

  const toLocal = (e: React.PointerEvent) => {
    const rect = svgRef.current?.getBoundingClientRect();
    if (!rect) return { x: 0, y: 0 };
    return { x: e.clientX - rect.left, y: e.clientY - rect.top };
  };

  const onPointerDown = (e: React.PointerEvent) => {
    drawing.current = true;
    last.current = toLocal(e);
  };

  const onPointerMove = (e: React.PointerEvent) => {
    if (!drawing.current || !last.current) return;
    const p = toLocal(e);
    const prev = last.current;
    setSegments((prevSeg) => [
      ...prevSeg,
      { color: strokeColor, x1: prev.x, y1: prev.y, x2: p.x, y2: p.y, width: strokeWidth },
    ]);
    last.current = p;
  };

  const onPointerUp = () => {
    drawing.current = false;
    last.current = null;
  };

  return (
    <div className="coloring">
      <div className="coloring-palette">
        {data.palette.map((c) => (
          <button
            key={c}
            className={`color-swatch ${!isEraser && color === c ? 'color-swatch-active' : ''}`}
            style={{ background: c }}
            onClick={() => {
              setIsEraser(false);
              setColor(c);
            }}
            aria-label={`颜色${c}`}
          />
        ))}
        <button
          className={`eraser-button ${isEraser ? 'eraser-active' : ''}`}
          onClick={() => setIsEraser(true)}
          aria-label="橡皮"
        >
          🧽
        </button>
        <button className="clear-button" onClick={() => setSegments([])} aria-label="清空">
          🗑️
        </button>
      </div>
      <svg
        ref={svgRef}
        className="coloring-canvas"
        viewBox="0 0 320 320"
        onPointerDown={onPointerDown}
        onPointerMove={onPointerMove}
        onPointerUp={onPointerUp}
        onPointerLeave={onPointerUp}
      >
        {segments.map((s, i) => (
          <line key={i} x1={s.x1} y1={s.y1} x2={s.x2} y2={s.y2} stroke={s.color} strokeWidth={s.width} strokeLinecap="round" />
        ))}
      </svg>
      <button className="big-button" onClick={onCorrect} aria-label="完成">
        完成
      </button>
    </div>
  );
}
```

```ts
let audioContext: AudioContext | null = null;

export function playTone(freq: number, durationMs = 400) {
  const Ctx = window.AudioContext ?? (window as unknown as { webkitAudioContext?: typeof AudioContext }).webkitAudioContext;
  if (!Ctx) return;
  if (!audioContext) audioContext = new Ctx();
  const ctx = audioContext;
  const osc = ctx.createOscillator();
  const gain = ctx.createGain();
  osc.type = 'sine';
  osc.frequency.value = freq;
  gain.gain.setValueAtTime(0.3, ctx.currentTime);
  gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + durationMs / 1000);
  osc.connect(gain);
  gain.connect(ctx.destination);
  osc.start();
  osc.stop(ctx.currentTime + durationMs / 1000);
}
```

```tsx
import { useCallback, useEffect, useRef, useState } from 'react';
import type { MusicData } from '../content/schema';
import { playTone } from './audio';

type Phase = 'idle' | 'playing' | 'listening';

interface Props {
  data: MusicData;
  onCorrect: () => void;
  onMistake: () => void;
}

export function MusicGame({ data, onCorrect, onMistake }: Props) {
  const [round, setRound] = useState(0);
  const [phase, setPhase] = useState<Phase>('idle');
  const [sequence, setSequence] = useState<number[]>([]);
  const [inputIndex, setInputIndex] = useState(0);
  const [activeNote, setActiveNote] = useState<number | null>(null);
  const [done, setDone] = useState(false);
  const timersRef = useRef<number[]>([]);

  const startRound = useCallback(
    (r: number) => {
      const seq = Array.from({ length: data.startLength + r }, () => Math.floor(Math.random() * data.notes.length));
      setSequence(seq);
      setInputIndex(0);
      setPhase('playing');
      seq.forEach((noteIdx, i) => {
        const t = window.setTimeout(() => {
          setActiveNote(noteIdx);
          playTone(data.notes[noteIdx].tone, 400);
          const clearT = window.setTimeout(() => setActiveNote(null), 350);
          timersRef.current.push(clearT);
        }, i * 600 + 300);
        timersRef.current.push(t);
      });
      const endT = window.setTimeout(() => setPhase('listening'), seq.length * 600 + 400);
      timersRef.current.push(endT);
    },
    [data],
  );

  const start = () => {
    setRound(0);
    setDone(false);
    startRound(0);
  };

  const handleTapNote = (noteIdx: number) => {
    if (done || phase !== 'listening') return;
    playTone(data.notes[noteIdx].tone, 400);
    if (noteIdx === sequence[inputIndex]) {
      const next = inputIndex + 1;
      if (next >= sequence.length) {
        if (round + 1 >= data.rounds) {
          setDone(true);
          onCorrect();
        } else {
          const nr = round + 1;
          setRound(nr);
          startRound(nr);
        }
      } else {
        setInputIndex(next);
      }
    } else {
      setInputIndex(0);
      onMistake();
      startRound(round);
    }
  };

  useEffect(() => () => timersRef.current.forEach((t) => window.clearTimeout(t)), []);

  return (
    <div className="music">
      <div className="music-status">
        {phase === 'idle' && (
          <button className="big-button" onClick={start} aria-label="开始">
            ▶️ 开始
          </button>
        )}
        {phase === 'playing' && <div className="music-listening">听一听…</div>}
        {phase === 'listening' && (
          <div className="music-listening">
            该你啦！第 {round + 1} / {data.rounds} 轮
          </div>
        )}
      </div>
      <div className="music-keys">
        {data.notes.map((n, idx) => (
          <button
            key={n.id}
            className={`music-key ${activeNote === idx ? 'music-key-active' : ''}`}
            onClick={() => handleTapNote(idx)}
            aria-label={n.id}
            disabled={phase !== 'listening'}
          >
            <span className="music-key-emoji">{n.emoji}</span>
          </button>
        ))}
      </div>
    </div>
  );
}
```

```ts
  await page.getByRole('button', { name: /艺术城堡/ }).click();

    await page.mouse.up();
```
