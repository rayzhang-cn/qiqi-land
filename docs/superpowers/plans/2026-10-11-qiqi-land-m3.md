```ts
export type ArithmeticOp = 'add' | 'sub' | 'mixed';

export interface ArithmeticConfig {
  op: ArithmeticOp;
  maxResult: number;
}

export interface Problem {
  a: number;
  b: number;
  op: 'add' | 'sub';
  answer: number;
  text: string;
}

function randInt(min: number, max: number, random: () => number): number {
  return min + Math.floor(random() * (max - min + 1));
}

export function generateProblem(config: ArithmeticConfig, random: () => number = Math.random): Problem {
  const op: 'add' | 'sub' = config.op === 'mixed' ? (random() < 0.5 ? 'add' : 'sub') : config.op;
  const max = config.maxResult;
  if (op === 'add') {
    const a = randInt(1, max - 1, random);
    const b = randInt(1, max - a, random);
    return { a, b, op, answer: a + b, text: `${a} + ${b}` };
  }
  const a = randInt(1, max, random);
  const b = randInt(0, a, random);
  return { a, b, op, answer: a - b, text: `${a} - ${b}` };
}

export function generateChoices(answer: number, max: number, random: () => number = Math.random): number[] {
  const choices = new Set<number>([answer]);
  let guard = 0;
  while (choices.size < 3 && guard < 100) {
    const delta = randInt(1, 3, random);
    const candidate = random() < 0.5 ? answer - delta : answer + delta;
    if (candidate >= 0 && candidate <= max && candidate !== answer) {
      choices.add(candidate);
    }
    guard++;
  }
  let fallback = 0;
  while (choices.size < 3) {
    if (!choices.has(fallback)) choices.add(fallback);
    fallback++;
  }
  return shuffle([...choices], random);
}

function shuffle<T>(arr: T[], random: () => number): T[] {
  const a = [...arr];
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(random() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}
```

```tsx
import { useState } from 'react';
import type { ArithmeticData } from '../content/schema';
import { generateProblem, generateChoices, type Problem } from './arithmetic';

interface Props {
  data: ArithmeticData;
  mode: 'choice' | 'fill';
  onCorrect: () => void;
  onMistake: () => void;
}

interface Round {
  problem: Problem;
  choices: number[];
}

export function ArithmeticGame({ data, mode, onCorrect, onMistake }: Props) {
  const [rounds] = useState<Round[]>(() =>
    Array.from({ length: data.problemCount }, () => {
      const problem = generateProblem(data);
      return { problem, choices: mode === 'choice' ? generateChoices(problem.answer, data.maxResult) : [] };
    }),
  );
  const [index, setIndex] = useState(0);
  const [input, setInput] = useState('');
  const [wrong, setWrong] = useState(false);

  const round = rounds[index];

  const handleCorrect = () => {
    if (index >= rounds.length - 1) {
      onCorrect();
    } else {
      setIndex(index + 1);
      setInput('');
      setWrong(false);
    }
  };

  const handleMistake = () => {
    setWrong(true);
    onMistake();
  };

  const pressDigit = (d: number) => {
    setWrong(false);
    setInput((prev) => (prev.length < 2 ? prev + d : prev));
  };
  const pressBackspace = () => setInput((prev) => prev.slice(0, -1));
  const pressConfirm = () => {
    if (input === '') return;
    if (parseInt(input, 10) === round.problem.answer) handleCorrect();
    else handleMistake();
  };

  return (
    <div className="arithmetic">
      <div className="arith-progress">
        第 {index + 1} / {rounds.length} 题
      </div>
      <div className="arith-question">{round.problem.text} = ?</div>
      {mode === 'choice' ? (
        <div className="choices-row">
          {round.choices.map((n) => (
            <button key={n} className="choice-button" onClick={() => (n === round.problem.answer ? handleCorrect() : handleMistake())}>
              {n}
            </button>
          ))}
        </div>
      ) : (
        <div className="fill-panel">
          <div className="fill-input" aria-label="答案">
            {input === '' ? '?' : input}
          </div>
          <div className="num-pad">
            {[1, 2, 3, 4, 5, 6, 7, 8, 9].map((d) => (
              <button key={d} className="num-key" onClick={() => pressDigit(d)}>
                {d}
              </button>
            ))}
            <button className="num-key" onClick={pressBackspace} aria-label="退格">
              ⌫
            </button>
            <button className="num-key" onClick={() => pressDigit(0)}>
              0
            </button>
            <button className="num-key num-confirm" onClick={pressConfirm} aria-label="确认">
              ✓
            </button>
          </div>
        </div>
      )}
      {wrong && <div className="arith-wrong">再试一次哦</div>}
    </div>
  );
}
```

```tsx
import { ProfileSelect } from './screens/ProfileSelect';

        />
      );
    }
    case 'poem-list':
      return (

          onSelectPoem={(poemId) => setScreen({ name: 'poem-recital', poemId })}
```

```tsx
import { useCallback, useEffect, useMemo, useRef, useState } from 'react';
import type { Poem } from '../content/schema';
import { toChars } from '../templates/poemText';
import { useSpeech } from '../engine/useSpeech';
import { useProfilesStore } from '../stores/profiles';
import { useProgressStore } from '../stores/progress';
import { Buddy } from '../components/Buddy';

export const MS_PER_CHAR = 350;

type Status = 'idle' | 'playing' | 'done';

interface CharItem {
  lineIndex: number;
  charIndex: number;
}

export function PoemRecital({ poem, onExit }: { poem: Poem; onExit: () => void }) {
  const { speak, cancel } = useSpeech();
  const activeId = useProfilesStore((s) => s.activeId);
  const avatar = useProfilesStore((s) => s.profiles.find((p) => p.id === s.activeId)?.avatar ?? '🦖');
  const recordAttempt = useProgressStore((s) => s.recordAttempt);

  // 朗读序列：诗名(0)、作者·朝代(1)、正文(2..)
  const reciteLines = useMemo(() => [poem.title, `${poem.author}，${poem.dynasty}`, ...poem.lines], [poem]);

  const items = useMemo<CharItem[]>(
    () => reciteLines.flatMap((line, lineIndex) => toChars(line).map((_, charIndex) => ({ lineIndex, charIndex }))),
    [reciteLines],
  );

  const [status, setStatus] = useState<Status>('idle');
  const [pos, setPos] = useState(-1);
  const timerRef = useRef<number | null>(null);

  const stopTimer = useCallback(() => {
    if (timerRef.current !== null) {
      window.clearInterval(timerRef.current);
      timerRef.current = null;
    }
  }, []);

  const start = useCallback(() => {
    stopTimer();
    setPos(0);
    setStatus('playing');
    timerRef.current = window.setInterval(() => {
      setPos((prev) => prev + 1);
    }, MS_PER_CHAR);
  }, [stopTimer]);

  const stop = useCallback(() => {
    stopTimer();
    cancel();
    setStatus('idle');
    setPos(-1);
  }, [stopTimer, cancel]);

  useEffect(() => {
    if (status !== 'playing') return;
    if (pos >= items.length) {
      stopTimer();
      setStatus('done');
      if (activeId) recordAttempt(activeId, poem.id, true, 1);
      return;
    }
    if (pos === 0) {
      speak(reciteLines[items[0].lineIndex]);
    } else if (items[pos].lineIndex !== items[pos - 1].lineIndex) {
      speak(reciteLines[items[pos].lineIndex]);
    }
  }, [pos, status, items, reciteLines, speak, stopTimer, activeId, recordAttempt]);

  useEffect(() => () => stopTimer(), [stopTimer]);

  if (status === 'done') {
    return (
      <div className="reward-view">
        <div className="reward-stars" aria-label="获得星星">
          ⭐
        </div>
        <Buddy avatar={avatar} text="背得真好！" />
        <button className="big-button" onClick={onExit}>
          继续
        </button>
      </div>
    );
  }

  const current = pos >= 0 && pos < items.length ? items[pos] : null;

  return (
    <div className="poem-screen">
      <div className="poem-title">
        <h2 className={current?.lineIndex === 0 ? 'poem-heading-reading' : ''}>{poem.title}</h2>
        <span className={`poem-author ${current?.lineIndex === 1 ? 'poem-author-reading' : ''}`}>
          {poem.author} · {poem.dynasty}
        </span>
      </div>
      <div className="poem-lines">
        {poem.lines.map((line, li) => (
          <div key={li} className="poem-line">
            {toChars(line).map((ch, ci) => (
              <span
                key={ci}
                className={current?.lineIndex === li + 2 && current?.charIndex === ci ? 'char char-current' : 'char'}
              >
                {ch}
              </span>
            ))}
          </div>
        ))}
      </div>
      <div className="poem-controls">
        {status === 'idle' && (
          <button className="poem-play" onClick={start} aria-label="开始听">
            ▶️ 开始听
          </button>
        )}
        {status === 'playing' && (
          <button className="poem-stop" onClick={stop} aria-label="停止">
            ⏹ 停止
          </button>
        )}
      </div>
    </div>
  );
}
```
